# Tests Tab Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A searchable Tests tab for the GM: the common non-combat tests (supply, panic, stopping
panic, survival) and environmental hazards, each with its roll, trigger, outcome, optional table
and rulebook page, plus a collapsed "How a test works" primer.

**Architecture:** Test content lives in a new `src/domain/tests.ts` (`TESTS`, `TEST_PRIMER`,
types), separate from `RULES`. Search and grouping are pure functions in
`src/domain/testSearch.ts`. A thin `src/ui/TestBrowser.tsx` renders them on new `#/tests` and
`#/tests/:id` routes. Nothing about characters changes.

**Tech Stack:** TypeScript, Preact, @preact/signals, Vitest. No new dependencies.

**Spec:** `docs/superpowers/specs/2026-10-13-tests-tab-design.md`

`0`→`Ń`); the success-die icon disappears, so "Za każdy wyrzucony ␣ …" means "for each success
rolled". Two columns are interleaved per line. When a number or condition is unclear, open the
PDF page with the Read tool (`pages: "108"`) and check the printed text.

### Wording rules

- `summary`: one sentence, ≤ 140 characters: what the test decides.
- `roll.pool`: the dice, terse. Skill tests as `Strength + Stamina` / `Siła + Kondycja`; other
  rolls as `Stress dice = supply rating (max 6)` / `Kości stresu = stan zapasów (maks. 6)`.
- `roll.canPush` / `roll.stressDice`: as the book states. If the book is silent, use the normal
  skill-test default (`true` / `true`) and add the entry to the PR notes.
- `when`: 1–2 sentences: the trigger and how often (round / turn / shift).
- `outcome`: 1–3 sentences: success, failure, escalation (e.g. to the Survival roll).
- Tables: short cells, our own words. No flavour text anywhere.
- Game terms in PL as in the book, lower-cased in prose (POZIOM STRESU → "poziom stresu", test
  KONDYCJI → "test Kondycji", runda / tura / wachta, Pokonany). In EN as in the English edition
  (Stress Level, Stamina roll, push, Broken, round / turn / shift).
- EN labels: no 2019 English rulebook is available. Use the plain English game term where it is
  certain (`Supply roll`, `Panic roll`); otherwise a descriptive name, listed in the PR notes for
  issue #31.
- PL and EN say the same thing.

```ts
import type { Localized } from './types'

export type TestId = string

export type TestGroup = 'core' | 'hazard'

export const TEST_GROUPS: readonly TestGroup[] = ['core', 'hazard']

export interface TestRoll {
  pool: Localized
  canPush: boolean
  stressDice: boolean
}

export interface TestTable {
  caption: Localized
  columns: Localized[]
  rows: Localized[][]
}

export interface TestEntry {
  id: TestId
  group: TestGroup
  label: Localized
  summary: Localized
  roll: TestRoll
  when: Localized
  outcome: Localized
  table?: TestTable
  related: TestId[]
  /** Printed rulebook page. */
  page: number
}

export interface PrimerSection {
  id: string
  label: Localized
  text: Localized
  page: number
}

/** Each group in book order. */
export const TESTS: readonly TestEntry[] = [
  {
    id: 'survival',
    group: 'core',
    page: 99,
    label: { en: 'Survival roll', pl: 'Test Przeżycia' },
    summary: {
      en: 'A Stamina roll that decides whether a dying character lives a little longer.',
      pl: 'Test Kondycji, który rozstrzyga, czy umierająca postać przeżyje jeszcze chwilę.',
    },
    roll: {
      pool: { en: 'Strength + Stamina', pl: 'Siła + Kondycja' },
      canPush: false,
      stressDice: false,
    },
    when: {
      en: 'When the time limit of a fatal critical injury runs out, and whenever a hazard calls for it while you are Broken.',
      pl: 'Gdy minie czas śmiertelnej rany ciężkiej oraz zawsze, gdy zagrożenie tego wymaga, a postać jest Pokonana.',
    },
    outcome: {
      en: 'Failure: the character dies. Success: they hold on, but roll again when the same time passes. First aid given before a failed roll can save them.',
      pl: 'Porażka: postać umiera. Sukces: trzyma się, ale po upływie tego samego czasu rzuca ponownie. Pierwsza pomoc udzielona przed nieudanym testem może ją ocalić.',
    },
    related: [],
  },
]

/** Book order. */
export const TEST_PRIMER: readonly PrimerSection[] = []

export function findTest(id: string): TestEntry | undefined {
  return TESTS.find((t) => t.id === id)
}

export function isTestId(id: string): boolean {
  return findTest(id) !== undefined
}

export function localizedFieldsOf(test: TestEntry): Localized[] {
  const fields = [test.label, test.summary, test.roll.pool, test.when, test.outcome]
  if (test.table) fields.push(test.table.caption, ...test.table.columns, ...test.table.rows.flat())
  return fields
}
```

  Medical Aid.
- `suffocation` (p. 110): out of air: Stamina roll every turn or after each exhausting action;
  modifier 0, −1, −2 … cumulative; failure: Health drops to 0 and a Survival roll every round
  until dead or back in breathable air. `related: ['survival', 'supply']`.

```ts
import type { TestEntry } from './tests'
import { filterTests, groupTests } from './testSearch'

const text = (en: string, pl: string) => ({ en, pl })
const blank = text('', '')

function entry(id: string, group: TestEntry['group'], extra: Partial<TestEntry> = {}): TestEntry {
  return {
    id,
    group,
    page: 1,
    label: text(id, id),
    summary: blank,
    roll: { pool: blank, canPush: true, stressDice: true },

    outcome: blank,
    related: [],

const fixture: TestEntry[] = [
```

```ts
import { TEST_GROUPS, localizedFieldsOf, type TestEntry, type TestGroup } from './tests'
import { matchesAllWords } from './textSearch'
import type { Lang } from './types'

export function filterTests(tests: readonly TestEntry[], query: string, lang: Lang): TestEntry[] {
  return tests.filter((t) =>
    matchesAllWords(
      query,
      localizedFieldsOf(t)
        .map((f) => f[lang])
        .join(' '),
    ),
  )
}

export function groupTests(
  tests: readonly TestEntry[],
): { group: TestGroup; tests: TestEntry[] }[] {
  return TEST_GROUPS.map((group) => ({
    group,
    tests: tests.filter((t) => t.group === group),
  })).filter((g) => g.tests.length > 0)
}
```

`src/ui/TestBrowser.tsx`:

```tsx
import { useEffect, useMemo, useState } from 'preact/hooks'
import {
  TESTS,
  TEST_PRIMER,
  findTest,
  type TestEntry,
  type TestId,
  type TestTable,
} from '../domain/tests'
import { filterTests, groupTests } from '../domain/testSearch'
import { lang, useLocalize, useT } from '../i18n'
import { href } from './router'

export function TestBrowser({ id }: { id?: TestId }) {
  const t = useT()
  const loc = useLocalize()
  const l = lang.value
  const [query, setQuery] = useState('')

  useEffect(() => {
    if (id) document.getElementById(entryId(id))?.scrollIntoView({ block: 'start' })
  }, [id])

  const groups = useMemo(() => groupTests(filterTests(TESTS, query, l)), [query, l])

  const entry = (test: TestEntry) => (
    <details key={test.id} id={entryId(test.id)} class="test" open={test.id === id}>
      <summary>
        <strong>{loc(test.label)}</strong>
        <span class="muted small">{t('talents.page', { page: test.page })}</span>
        <span class="small talent-summary">{loc(test.summary)}</span>
      </summary>
      <dl>
        <dt>{t('tests.roll')}</dt>
        <dd>
          {loc(test.roll.pool)} · {t(test.roll.canPush ? 'tests.push.yes' : 'tests.push.no')} ·{' '}
          {t(test.roll.stressDice ? 'tests.stress.yes' : 'tests.stress.no')}
        </dd>
        <dt>{t('tests.when')}</dt>
        <dd>{loc(test.when)}</dd>
        <dt>{t('tests.outcome')}</dt>
        <dd>{loc(test.outcome)}</dd>
      </dl>
      {test.table && <TableView table={test.table} />}
      {test.related.length > 0 && (
        <p class="test-related small">
          {t('tests.related')}
          {test.related.map((rid) => {
            const target = findTest(rid)
            return target ? (
              <a key={rid} href={href.tests(rid)}>
                {loc(target.label)}
              </a>
            ) : null
          })}
        </p>
      )}
    </details>
  )

  return (
    <section class="tests">
      <div class="toolbar">
        <h2>{t('tests.title')}</h2>
      </div>
      <div class="filters">
        <input
          type="search"
          value={query}
          placeholder={t('tests.search')}
          aria-label={t('tests.search')}
          onInput={(e) => setQuery(e.currentTarget.value)}
        />
      </div>

      <details class="panel primer">
        <summary>{t('tests.primer')}</summary>
        {TEST_PRIMER.map((s) => (
          <section key={s.id}>
            <h4>
              {loc(s.label)} <span class="muted small">{t('talents.page', { page: s.page })}</span>
            </h4>
            <p>{loc(s.text)}</p>
          </section>
        ))}
      </details>

      {groups.length === 0 && <p class="empty muted">{t('tests.empty')}</p>}

      {groups.map((g) => (
        <section key={g.group} class="panel">
          <h3>{t(g.group === 'core' ? 'tests.group.core' : 'tests.group.hazard')}</h3>
          {g.tests.map(entry)}
        </section>
      ))}
    </section>
  )
}

function TableView({ table }: { table: TestTable }) {
  const loc = useLocalize()
  return (
    <div class="test-table">
      <table>
        <caption>{loc(table.caption)}</caption>
        <thead>
          <tr>
            {table.columns.map((c, i) => (
              <th key={i} scope="col">
                {loc(c)}
              </th>
            ))}
          </tr>
        </thead>
        <tbody>
          {table.rows.map((row, r) => (
            <tr key={r}>
              {row.map((cell, c) => (
                <td key={c}>{loc(cell)}</td>
              ))}
            </tr>
          ))}
        </tbody>
      </table>
    </div>
  )
}

function entryId(id: TestId): string {
  return `test-${id}`
}
```
